Module: Two DC motors on an H-bridge (id: dual_hbridge, category: motor, complexity: simple)
Pins:
- 8 OUTPUT: motor A IN1
- 9 OUTPUT: motor A IN2
- 6 OUTPUT: motor A enable (PWM)
- 10 OUTPUT: motor B IN3
- 11 OUTPUT: motor B IN4
- 5 OUTPUT: motor B enable (PWM)
Functions to test:
- void hb_init()
- bool hb_forward(uint8_t motor, uint8_t speed)
- bool hb_backward(uint8_t motor, uint8_t speed)
- bool hb_stop(uint8_t motor)
- void hb_stop_all()
- int8_t hb_direction(uint8_t motor)
- uint8_t hb_speed(uint8_t motor)

### logic.h
```cpp
// dual_hbridge: two DC motors on an L298N/L293D H-bridge (IN1..IN4 direction, ENA/ENB PWM speed).
// Adapted from https://github.com/Projectslearner/arduino-uno-h-bridge-module (MIT).
#ifndef LOGIC_H
#define LOGIC_H
#include <Arduino.h>

#define HB_MOTOR_A 0
#define HB_MOTOR_B 1

#define HB_PIN_IN1 8    // motor A direction
#define HB_PIN_IN2 9
#define HB_PIN_ENA 6    // motor A speed (PWM)
#define HB_PIN_IN3 10   // motor B direction
#define HB_PIN_IN4 11
#define HB_PIN_ENB 5    // motor B speed (PWM)

void hb_init();                                   // pins OUTPUT, both motors stopped
bool hb_forward(uint8_t motor, uint8_t speed);    // INx HIGH / INy LOW, EN = speed; false if motor invalid
bool hb_backward(uint8_t motor, uint8_t speed);   // INx LOW / INy HIGH, EN = speed
bool hb_stop(uint8_t motor);                      // both IN LOW, EN 0
void hb_stop_all();
int8_t hb_direction(uint8_t motor);               // +1 forward, -1 backward, 0 stopped (or invalid motor)
uint8_t hb_speed(uint8_t motor);                  // last PWM speed set (0 if stopped)

#endif
```

### logic.cpp
```cpp
#include "logic.h"

static int8_t g_dir[2] = {0, 0};
static uint8_t g_speed[2] = {0, 0};

static bool hb_pins(uint8_t motor, uint8_t &in_a, uint8_t &in_b, uint8_t &en) {
  if (motor == HB_MOTOR_A) {
    in_a = HB_PIN_IN1;
    in_b = HB_PIN_IN2;
    en = HB_PIN_ENA;
    return true;
  }
  if (motor == HB_MOTOR_B) {
    in_a = HB_PIN_IN3;
    in_b = HB_PIN_IN4;
    en = HB_PIN_ENB;
    return true;
  }
  return false;
}

void hb_init() {
  pinMode(HB_PIN_IN1, OUTPUT);
  pinMode(HB_PIN_IN2, OUTPUT);
  pinMode(HB_PIN_ENA, OUTPUT);
  pinMode(HB_PIN_IN3, OUTPUT);
  pinMode(HB_PIN_IN4, OUTPUT);
  pinMode(HB_PIN_ENB, OUTPUT);
  hb_stop_all();
}

bool hb_forward(uint8_t motor, uint8_t speed) {
  uint8_t a, b, en;
  if (!hb_pins(motor, a, b, en)) {
    return false;
  }
  digitalWrite(a, HIGH);
  digitalWrite(b, LOW);
  analogWrite(en, speed);
  g_dir[motor] = 1;
  g_speed[motor] = speed;
  return true;
}

bool hb_backward(uint8_t motor, uint8_t speed) {
  uint8_t a, b, en;
  if (!hb_pins(motor, a, b, en)) {
    return false;
  }
  digitalWrite(a, LOW);
  digitalWrite(b, HIGH);
  analogWrite(en, speed);
  g_dir[motor] = -1;
  g_speed[motor] = speed;
  return true;
}

bool hb_stop(uint8_t motor) {
  uint8_t a, b, en;
  if (!hb_pins(motor, a, b, en)) {
    return false;
  }
  digitalWrite(a, LOW);
  digitalWrite(b, LOW);
  analogWrite(en, 0);
  g_dir[motor] = 0;
  g_speed[motor] = 0;
  return true;
}

void hb_stop_all() {
  hb_stop(HB_MOTOR_A);
  hb_stop(HB_MOTOR_B);
}

int8_t hb_direction(uint8_t motor) {
  if (motor > HB_MOTOR_B) {
    return 0;
  }
  return g_dir[motor];
}

uint8_t hb_speed(uint8_t motor) {
  if (motor > HB_MOTOR_B) {
    return 0;
  }
  return g_speed[motor];
}
```

## Covering a function
- Call the module's init function at the start of every test so that tests do not depend on each other.
- For every output a function drives, assert each pin it should set and each pin it should leave unchanged.
- For functions with an index or a range argument (LED number, state number), test the first value, the last
  value and one value outside the range.

[kb/guides/testing_digital_pins.md]
Testing digital pins on the Arduino Uno / Pin numbers on the Uno

## Pin numbers on the Uno
- Digital pins 0 to 13. The analog inputs A0 to A5 are also digital pins 14 to 19.
- Pins 0 and 1 are the Serial RX/TX lines; the test protocol uses Serial, so tests must not drive them.
- PWM pins: 3, 5, 6, 9, 10, 11. `LED_BUILTIN` is pin 13.
- Check outputs against the pin numbers in the module's specification (its pin table), not against constants
  taken from the code under test: a wrong pin number in the code is only detected if the test uses the pin
  number the hardware is actually wired to.

# Arduino and Stepper Motor Configurations

Stepper motors, due to their unique design, can be controlled to a high degree of accuracy without any feedback mechanisms. The shaft of a stepper, mounted with a series of magnets, is controlled by a series of electromagnetic coils that are charged positively and negatively in a specific sequence, precisely moving it forward or backward in small "steps".

There are two types of steppers, Unipolars and Bipolars, and it is very important to know which type you are working with. For each of the motors, there is a different circuit. The example code will control both kinds of motors. See the  [unipolar](https://www.arduino.cc/en/Reference/StepperUnipolarCircuit) and [bipolar](https://www.arduino.cc/en/Reference/StepperBipolarCircuit) motor schematics for information on how to wire up your motor.

The stepper is controlled by with digital pins 8, 9, 10, and 11 for either unipolar or bipolar motors. The Arduino board will connect to a [U2004 Darlington Array](http://octopart.com/uln2004a-stmicroelectronics-4798) if you're using a unipolar stepper or a [SN754410NE H-Bridge](http://octopart.com/sn754410ne-texas+instruments-1320) if you have a bipolar motor.

```arduino
int directionPin = 12;
int pwmPin = 3;
int brakePin = 9;

//uncomment if using channel B, and remove above definitions
//int directionPin = 13;
//int pwmPin = 11;
//int brakePin = 8;

//boolean to switch direction
bool directionState;

void setup() {

//define pins
pinMode(directionPin, OUTPUT);
pinMode(pwmPin, OUTPUT);
pinMode(brakePin, OUTPUT);

}

void loop() {

//change direction every loop()
directionState = !directionState;

//write a low state to the direction pin (13)
if(directionState == false){
  digitalWrite(directionPin, LOW);
}

//write a high state to the direction pin (13)
else{
  digitalWrite(directionPin, HIGH);
}

//release breaks
digitalWrite(brakePin, LOW);

//set work duty for the motor
analogWrite(pwmPin, 30);

delay(2000);

//activate breaks
digitalWrite(brakePin, HIGH);

//set work duty for the motor to 0 (off)
analogWrite(pwmPin, 0);

delay(2000);
}
```

## Programming the Board

We will now get to the programming part of this tutorial.

First, let's take a look at some key commands in the code. We are actually not using a library, as the operation is very basic.

- `int directionPin = 12;` - assign direction pin.
- `int pwmPin = 3;` - assign PWM (work duty) pin.
- `int brakePin = 9;` - assign brake pin.
- `digitalWrite(directionPin, state)` - sets the direction of the pin by using HIGH or LOW states.
- `digitalWrite(brakePin, state)` - release or activate brakes, using HIGH or LOW states.
- `analogWrite(pwmPin, 30)` - write a value between 0-100 to set the work duty.
- `directionState = !directionState` - a boolean that switches every time the loop is run.
