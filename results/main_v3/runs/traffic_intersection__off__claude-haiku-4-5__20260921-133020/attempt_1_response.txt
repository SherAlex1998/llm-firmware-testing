#include "logic.h"
#include "test_runner.h"

TEST(init_all_yellow) {
  ti_init();
  ASSERT_EQ(ti_direction_color(0), TI_YELLOW);
  ASSERT_EQ(ti_direction_color(1), TI_YELLOW);
  ASSERT_EQ(ti_direction_color(2), TI_YELLOW);
  ASSERT_EQ(ti_direction_color(3), TI_YELLOW);
  ASSERT_PIN(13, LOW);
  ASSERT_PIN(12, LOW);
  ASSERT_PIN(11, HIGH);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, LOW);
  ASSERT_PIN(8, HIGH);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, HIGH);
  ASSERT_PIN(4, LOW);
  ASSERT_PIN(3, LOW);
  ASSERT_PIN(2, HIGH);
}

TEST(set_direction_valid) {
  ti_init();
  ASSERT_TRUE(ti_set_direction(0, TI_RED));
  ASSERT_EQ(ti_direction_color(0), TI_RED);
  ASSERT_PIN(13, HIGH);
  ASSERT_PIN(12, LOW);
  ASSERT_PIN(11, LOW);
  
  ASSERT_TRUE(ti_set_direction(1, TI_GREEN));
  ASSERT_EQ(ti_direction_color(1), TI_GREEN);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, HIGH);
  ASSERT_PIN(8, LOW);
  
  ASSERT_TRUE(ti_set_direction(2, TI_YELLOW));
  ASSERT_EQ(ti_direction_color(2), TI_YELLOW);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, HIGH);
}

TEST(set_direction_invalid) {
  ti_init();
  ASSERT_FALSE(ti_set_direction(4, TI_RED));
  ASSERT_FALSE(ti_set_direction(0, 3));
  ASSERT_FALSE(ti_set_direction(5, TI_GREEN));
}

TEST(apply_phase_green) {
  ti_init();
  ASSERT_TRUE(ti_apply_phase(0));
  ASSERT_EQ(ti_phase(), 0);
  ASSERT_EQ(ti_direction_color(0), TI_GREEN);
  ASSERT_EQ(ti_direction_color(1), TI_RED);
  ASSERT_EQ(ti_direction_color(2), TI_RED);
  ASSERT_EQ(ti_direction_color(3), TI_RED);
  ASSERT_PIN(12, HIGH);
  ASSERT_PIN(13, LOW);
  ASSERT_PIN(10, HIGH);
  ASSERT_PIN(9, LOW);
}

TEST(apply_phase_yellow) {
  ti_init();
  ASSERT_TRUE(ti_apply_phase(1));
  ASSERT_EQ(ti_phase(), 1);
  ASSERT_EQ(ti_direction_color(0), TI_YELLOW);
  ASSERT_EQ(ti_direction_color(1), TI_RED);
  ASSERT_EQ(ti_direction_color(2), TI_RED);
  ASSERT_EQ(ti_direction_color(3), TI_RED);
  ASSERT_PIN(11, HIGH);
  ASSERT_PIN(13, LOW);
}

TEST(apply_phase_invalid) {
  ti_init();
  ASSERT_FALSE(ti_apply_phase(8));
  ASSERT_FALSE(ti_apply_phase(9));
  ASSERT_EQ(ti_phase(), 0);
}

TEST(next_phase_sequence) {
  ti_init();
  ti_apply_phase(0);
  ASSERT_EQ(ti_phase(), 0);
  
  ti_next_phase();
  ASSERT_EQ(ti_phase(), 1);
  ASSERT_EQ(ti_direction_color(0), TI_YELLOW);
  
  ti_next_phase();
  ASSERT_EQ(ti_phase(), 2);
  ASSERT_EQ(ti_direction_color(1), TI_GREEN);
  
  ti_next_phase();
  ti_next_phase();
  ti_next_phase();
  ti_next_phase();
  ti_next_phase();
  ASSERT_EQ(ti_phase(), 7);
  
  ti_next_phase();
  ASSERT_EQ(ti_phase(), 0);
}

TEST(is_safe_conditions) {
  ti_init();
  ASSERT_FALSE(ti_is_safe());
  
  ti_apply_phase(0);
  ASSERT_TRUE(ti_is_safe());
  
  ti_apply_phase(1);
  ASSERT_TRUE(ti_is_safe());
  
  ti_apply_phase(4);
  ASSERT_TRUE(ti_is_safe());
  
  ti_set_direction(0, TI_GREEN);
  ti_set_direction(1, TI_GREEN);
  ASSERT_FALSE(ti_is_safe());
  
  ti_set_direction(1, TI_RED);
  ASSERT_TRUE(ti_is_safe());
}

void setup() {
  Serial.begin(9600);
  RUN_ALL_TESTS();
}

void loop() {}
