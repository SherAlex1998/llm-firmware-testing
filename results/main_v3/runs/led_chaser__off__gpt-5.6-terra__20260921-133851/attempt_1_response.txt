#include "logic.h"
#include "test_runner.h"

TEST(init_sets_all_leds_off_and_forward) {
  chaser_init();
  ASSERT_EQ(chaser_position(), -1);
  ASSERT_EQ(chaser_direction(), 1);
  ASSERT_EQ(chaser_count_on(), 0);
  ASSERT_PIN(2, LOW);
  ASSERT_PIN(3, LOW);
  ASSERT_PIN(4, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(8, LOW);
  ASSERT_PIN(9, LOW);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(11, LOW);
}

TEST(light_only_selects_requested_led) {
  chaser_init();
  ASSERT_TRUE(chaser_light_only(4));
  ASSERT_EQ(chaser_position(), 4);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(2, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(6, HIGH);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(11, LOW);
}

TEST(light_only_rejects_out_of_range_without_change) {
  chaser_init();
  ASSERT_TRUE(chaser_light_only(7));
  ASSERT_FALSE(chaser_light_only(10));
  ASSERT_EQ(chaser_position(), 7);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(9, HIGH);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(11, LOW);
}

TEST(all_off_clears_selected_led_and_position) {
  chaser_init();
  ASSERT_TRUE(chaser_light_only(2));
  chaser_all_off();
  ASSERT_EQ(chaser_position(), -1);
  ASSERT_EQ(chaser_count_on(), 0);
  ASSERT_PIN(2, LOW);
  ASSERT_PIN(4, LOW);
  ASSERT_PIN(11, LOW);
}

TEST(step_moves_forward_from_off_and_middle) {
  chaser_init();
  chaser_step();
  ASSERT_EQ(chaser_position(), 0);
  ASSERT_EQ(chaser_direction(), 1);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(2, HIGH);

  chaser_step();
  ASSERT_EQ(chaser_position(), 1);
  ASSERT_EQ(chaser_direction(), 1);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(2, LOW);
  ASSERT_PIN(3, HIGH);
}

TEST(step_bounces_at_both_ends) {
  chaser_init();
  ASSERT_TRUE(chaser_light_only(9));
  chaser_step();
  ASSERT_EQ(chaser_position(), 8);
  ASSERT_EQ(chaser_direction(), -1);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(10, HIGH);
  ASSERT_PIN(11, LOW);

  ASSERT_TRUE(chaser_light_only(0));
  chaser_step();
  ASSERT_EQ(chaser_position(), 1);
  ASSERT_EQ(chaser_direction(), 1);
  ASSERT_EQ(chaser_count_on(), 1);
  ASSERT_PIN(2, LOW);
  ASSERT_PIN(3, HIGH);
}

void setup() { Serial.begin(9600); RUN_ALL_TESTS(); }
void loop() {}
