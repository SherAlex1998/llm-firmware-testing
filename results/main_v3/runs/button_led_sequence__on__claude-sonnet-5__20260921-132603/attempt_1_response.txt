#include "logic.h"
#include "test_runner.h"

TEST(init_sets_all_leds_off) {
  seq_init();
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
  ASSERT_EQ(seq_count(), 0);
  ASSERT_EQ(seq_current_led(), -1);
}

TEST(apply_step_out_of_range) {
  seq_init();
  ASSERT_TRUE(seq_apply(0));
  ASSERT_FALSE(seq_apply(4));
  ASSERT_FALSE(seq_apply(255));
  ASSERT_EQ(seq_current_led(), 0);
  ASSERT_PIN(7, HIGH);
}

TEST(apply_sequential_steps) {
  seq_init();
  ASSERT_TRUE(seq_apply(0));
  ASSERT_PIN(7, HIGH);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);

  ASSERT_TRUE(seq_apply(1));
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, HIGH);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);

  ASSERT_TRUE(seq_apply(2));
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, HIGH);

  ASSERT_TRUE(seq_apply(3));
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, HIGH);
  ASSERT_EQ(seq_current_led(), 3);
}

TEST(advance_cycles_count_and_leds) {
  seq_init();
  ASSERT_EQ(seq_count(), 0);

  seq_advance(); // applies step 0, count -> 1
  ASSERT_PIN(7, HIGH);
  ASSERT_EQ(seq_count(), 1);
  ASSERT_EQ(seq_current_led(), 0);

  seq_advance(); // applies step 1, count -> 2
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, HIGH);
  ASSERT_EQ(seq_count(), 2);

  seq_advance(); // applies step 2, count -> 3
  seq_advance(); // applies step 3, count -> 0 (wrap)
  ASSERT_EQ(seq_count(), 0);
  ASSERT_PIN(4, HIGH);
  ASSERT_EQ(seq_current_led(), 3);
}

TEST(all_off_clears_leds_keeps_count) {
  seq_init();
  seq_advance(); // count -> 1, led0 on
  seq_all_off();
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
  ASSERT_EQ(seq_current_led(), -1);
  ASSERT_EQ(seq_count(), 1);
}

TEST(button_pressed_reflects_pin) {
  seq_init();
  pinMode(SEQ_PIN_BUTTON, OUTPUT);
  digitalWrite(SEQ_PIN_BUTTON, LOW);
  ASSERT_FALSE(seq_button_pressed());
  digitalWrite(SEQ_PIN_BUTTON, HIGH);
  ASSERT_TRUE(seq_button_pressed());
  pinMode(SEQ_PIN_BUTTON, INPUT);
}

TEST(poll_advances_only_when_pressed) {
  seq_init();
  pinMode(SEQ_PIN_BUTTON, OUTPUT);
  digitalWrite(SEQ_PIN_BUTTON, LOW);
  seq_poll();
  ASSERT_EQ(seq_count(), 0);
  ASSERT_EQ(seq_current_led(), -1);

  digitalWrite(SEQ_PIN_BUTTON, HIGH);
  seq_poll();
  ASSERT_EQ(seq_count(), 1);
  ASSERT_PIN(7, HIGH);
  ASSERT_EQ(seq_current_led(), 0);

  pinMode(SEQ_PIN_BUTTON, INPUT);
}

void setup() { Serial.begin(9600); RUN_ALL_TESTS(); }
void loop() {}
