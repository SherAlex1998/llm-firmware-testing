```cpp
// joystick_motor: analog joystick axis -> DC motor speed and direction through an L293D.
// Adapted from https://github.com/vchervanev/arduino-joystick-to-motor (MIT).
#ifndef LOGIC_H
#define LOGIC_H
#include <Arduino.h>

#define JM_PIN_AXIS   A0   // joystick X axis (analog)
#define JM_PIN_SPEED  5    // L293D enable (PWM)
#define JM_PIN_DIR_A  3    // L293D input A
#define JM_PIN_DIR_B  4    // L293D input B
#define JM_AXIS_CENTER 512
#define JM_DEAD_ZONE   10  // speeds below this stop the motor

void jm_init();                          // pins configured, clockwise, stopped
void jm_set_direction(bool clockwise);   // clockwise: DIR_A HIGH / DIR_B LOW; otherwise swapped
bool jm_direction();                     // true = clockwise
void jm_set_speed(int speed);            // clamped to 0..255, written to the PWM pin
int jm_speed();                          // last speed written
void jm_stop();                          // speed 0
int jm_speed_from_axis(int dx);          // pure: |dx - 512| / 2
bool jm_clockwise_from_axis(int dx);     // pure: dx > 512
void jm_apply_axis(int dx);              // dead zone -> stop, else set direction and speed
void jm_update();                        // read the joystick and apply

#endif
```

### logic.cpp
```cpp
#include "logic.h"

static bool g_clockwise = true;
static int g_speed = 0;

void jm_init() {
  pinMode(JM_PIN_AXIS, INPUT);
  pinMode(JM_PIN_SPEED, OUTPUT);
  pinMode(JM_PIN_DIR_A, OUTPUT);
  pinMode(JM_PIN_DIR_B, OUTPUT);
  jm_set_direction(true);
  jm_stop();
}

void jm_set_direction(bool clockwise) {
  g_clockwise = clockwise;
  if (clockwise) {
    digitalWrite(JM_PIN_DIR_A, HIGH);
    digitalWrite(JM_PIN_DIR_B, LOW);
  } else {
    digitalWrite(JM_PIN_DIR_A, LOW);
    digitalWrite(JM_PIN_DIR_B, HIGH);
  }
}

bool jm_direction() {
  return g_clockwise;
}

void jm_set_speed(int speed) {
  if (speed < 0) {
    speed = 0;
  }
  if (speed > 255) {
    speed = 255;
  }
  g_speed = speed;
  analogWrite(JM_PIN_SPEED, g_speed);
}

int jm_speed() {
  return g_speed;
}

void jm_stop() {
  jm_set_speed(0);
}

int jm_speed_from_axis(int dx) {
  return abs(dx - JM_AXIS_CENTER) / 2;   // ~0..256
}

bool jm_clockwise_from_axis(int dx) {
  return dx > JM_AXIS_CENTER;
}

void jm_apply_axis(int dx) {
  int speed = jm_speed_from_axis(dx);
  if (speed < JM_DEAD_ZONE) {
    jm_stop();
  } else {
    jm_set_direction(jm_clockwise_from_axis(dx));
    jm_set_speed(speed);
  }
}

void jm_update() {
  jm_apply_axis(analogRead(JM_PIN_AXIS));
}
```

## Programming the Board

We will now get to the programming part of this tutorial.

First, let's take a look at some key commands in the code. We are actually not using a library, as the operation is very basic.

- `int directionPin = 12;` - assign direction pin.
- `int pwmPin = 3;` - assign PWM (work duty) pin.
- `int brakePin = 9;` - assign brake pin.
- `digitalWrite(directionPin, state)` - sets the direction of the pin by using HIGH or LOW states.
- `digitalWrite(brakePin, state)` - release or activate brakes, using HIGH or LOW states.
- `analogWrite(pwmPin, 30)` - write a value between 0-100 to set the work duty.
- `directionState = !directionState` - a boolean that switches every time the loop is run.

The sketch can be found in the snippet below. Upload the sketch to the board.

Testing digital pins on the Arduino Uno / Pin numbers on the Uno

## Pin numbers on the Uno
- Digital pins 0 to 13. The analog inputs A0 to A5 are also digital pins 14 to 19.
- Pins 0 and 1 are the Serial RX/TX lines; the test protocol uses Serial, so tests must not drive them.
- PWM pins: 3, 5, 6, 9, 10, 11. `LED_BUILTIN` is pin 13.
- Check outputs against the pin numbers in the module's specification (its pin table), not against constants
  taken from the code under test: a wrong pin number in the code is only detected if the test uses the pin
  number the hardware is actually wired to.

[kb/guides/testing_digital_pins.md]
Testing digital pins on the Arduino Uno / Covering a function

## Covering a function
- Call the module's init function at the start of every test so that tests do not depend on each other.
- For every output a function drives, assert each pin it should set and each pin it should leave unchanged.
- For functions with an index or a range argument (LED number, state number), test the first value, the last
  value and one value outside the range.

# Testing PWM, analog inputs and servos

## analogWrite and what a test can observe
`analogWrite(pin, value)` sets the pin to `OUTPUT`. `analogWrite(pin, 0)` is implemented as
`digitalWrite(pin, LOW)` and `analogWrite(pin, 255)` as `digitalWrite(pin, HIGH)`; any other value starts the
PWM timer (Arduino AVR core, `wiring_analog.c`). On the Uno the PWM frequency is 490 Hz, and 980 Hz on pins 5
and 6.

Only the two extreme values can be checked with `digitalRead()`. For values 1 to 254 the duty cycle cannot be
read back, and `digitalRead()` would switch the PWM off. Check intermediate levels through the module's own
state: a getter that returns the level it wrote, or the value returned by the function that computes the level.
