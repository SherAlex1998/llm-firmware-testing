#include "logic.h"
#include "test_runner.h"

TEST(init_state) {
  seq_init();
  ASSERT_EQ(seq_count(), 0);
  ASSERT_EQ(seq_current_led(), -1);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
}

TEST(apply_valid_step) {
  seq_init();
  bool result = seq_apply(0);
  ASSERT_TRUE(result);
  ASSERT_EQ(seq_current_led(), 0);
  ASSERT_PIN(7, HIGH);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
}

TEST(apply_out_of_range) {
  seq_init();
  bool result = seq_apply(4);
  ASSERT_FALSE(result);
  ASSERT_EQ(seq_current_led(), -1);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
}

TEST(advance_sequence) {
  seq_init();
  seq_advance();
  ASSERT_EQ(seq_count(), 1);
  ASSERT_EQ(seq_current_led(), 0);
  ASSERT_PIN(7, HIGH);
  
  seq_advance();
  ASSERT_EQ(seq_count(), 2);
  ASSERT_EQ(seq_current_led(), 1);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, HIGH);
  
  seq_advance();
  ASSERT_EQ(seq_count(), 3);
  ASSERT_EQ(seq_current_led(), 2);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, HIGH);
  
  seq_advance();
  ASSERT_EQ(seq_count(), 0);
  ASSERT_EQ(seq_current_led(), 3);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, HIGH);
}

TEST(all_off) {
  seq_init();
  seq_advance();
  seq_advance();
  ASSERT_EQ(seq_current_led(), 1);
  seq_all_off();
  ASSERT_EQ(seq_current_led(), -1);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, LOW);
  ASSERT_PIN(4, LOW);
}

TEST(button_not_pressed) {
  seq_init();
  pinMode(SEQ_PIN_BUTTON, INPUT);
  digitalWrite(SEQ_PIN_BUTTON, LOW);
  bool pressed = seq_button_pressed();
  ASSERT_FALSE(pressed);
}

TEST(poll_no_button) {
  seq_init();
  pinMode(SEQ_PIN_BUTTON, INPUT);
  digitalWrite(SEQ_PIN_BUTTON, LOW);
  uint8_t initial_count = seq_count();
  seq_poll();
  ASSERT_EQ(seq_count(), initial_count);
}

void setup() {
  Serial.begin(9600);
  RUN_ALL_TESTS();
}

void loop() {}
