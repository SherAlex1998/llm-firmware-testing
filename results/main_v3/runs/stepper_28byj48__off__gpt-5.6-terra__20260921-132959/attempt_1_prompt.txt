```cpp
// stepper_28byj48: 28BYJ-48 stepper on a ULN2003 driver, full-step and half-step sequencing on 4 pins.
// Adapted from https://github.com/Moragor/Mora_28BYJ_48 (MIT); class converted to free functions.
#ifndef LOGIC_H
#define LOGIC_H
#include <Arduino.h>

#define STEPPER_PIN_IN1 8
#define STEPPER_PIN_IN2 9
#define STEPPER_PIN_IN3 10
#define STEPPER_PIN_IN4 11
#define STEPPER_SEQUENCE_LENGTH 8
#define STEPPER_DEFAULT_DELAY_MS 2

void stepper_init();                          // pins OUTPUT, coils off, phase 0, position 0
void stepper_set_delay(unsigned int ms);      // delay after each step
unsigned int stepper_delay();
uint8_t stepper_coil_pattern(uint8_t phase);  // pure: 4-bit coil pattern for phase 0..7 (bit0 = IN1); 0 if invalid
bool stepper_apply_phase(uint8_t phase);      // write the coil pattern of a phase to the pins; false if invalid
int8_t stepper_phase();                       // current phase index 0..7
long stepper_position();                      // signed position in half-steps (full step = 2)
void stepper_step_full(int steps);            // >0 forward, <0 backward; uses the two-coil phases (1,3,5,7)
void stepper_step_half(int steps);            // >0 forward, <0 backward; walks all 8 phases
void stepper_release();                       // all coils off (phase/position unchanged)

#endif
```

### logic.cpp
```cpp
#include "logic.h"

// Coil energising order for one electrical cycle (bit0 = IN1 ... bit3 = IN4).
static const uint8_t k_steps[STEPPER_SEQUENCE_LENGTH] = {
  B1000, B1100, B0100, B0110, B0010, B0011, B0001, B1001
};

static int8_t g_phase = 0;
static long g_position = 0;
static unsigned int g_delay_ms = STEPPER_DEFAULT_DELAY_MS;

void stepper_init() {
  pinMode(STEPPER_PIN_IN1, OUTPUT);
  pinMode(STEPPER_PIN_IN2, OUTPUT);
  pinMode(STEPPER_PIN_IN3, OUTPUT);
  pinMode(STEPPER_PIN_IN4, OUTPUT);
  g_phase = 0;
  g_position = 0;
  g_delay_ms = STEPPER_DEFAULT_DELAY_MS;
  stepper_release();
}

void stepper_set_delay(unsigned int ms) {
  g_delay_ms = ms;
}

unsigned int stepper_delay() {
  return g_delay_ms;
}

uint8_t stepper_coil_pattern(uint8_t phase) {
  if (phase >= STEPPER_SEQUENCE_LENGTH) {
    return 0;
  }
  return k_steps[phase];
}

bool stepper_apply_phase(uint8_t phase) {
  if (phase >= STEPPER_SEQUENCE_LENGTH) {
    return false;
  }
  uint8_t pattern = k_steps[phase];
  digitalWrite(STEPPER_PIN_IN1, bitRead(pattern, 0));
  digitalWrite(STEPPER_PIN_IN2, bitRead(pattern, 1));
  digitalWrite(STEPPER_PIN_IN3, bitRead(pattern, 2));
  digitalWrite(STEPPER_PIN_IN4, bitRead(pattern, 3));
  g_phase = (int8_t)phase;
  return true;
}

int8_t stepper_phase() {
  return g_phase;
}

long stepper_position() {
  return g_position;
}

void stepper_step_full(int steps) {
  while (steps > 0) {
    if (g_phase % 2 != 1) {
      g_phase++;          // move onto an odd (two-coil) phase
    } else {
      g_phase += 2;       // advance two positions per full step
    }
    if (g_phase > 7) {
      g_phase = 1;
    }
    stepper_apply_phase(g_phase);
    delay(g_delay_ms);
    g_position += 2;
    steps--;
  }
  while (steps < 0) {
    if (g_phase % 2 != 1) {
      g_phase--;
    } else {
      g_phase -= 2;
    }
    if (g_phase < 1) {
      g_phase = 7;
    }
    stepper_apply_phase(g_phase);
    delay(g_delay_ms);
    g_position -= 2;
    steps++;
  }
}

void stepper_step_half(int steps) {
  while (steps > 0) {
    g_phase++;
    if (g_phase > 7) {
      g_phase = 0;
    }
    stepper_apply_phase(g_phase);
    delay(g_delay_ms);
    g_position++;
    steps--;
  }
  while (steps < 0) {
    g_phase--;
    if (g_phase < 0) {
      g_phase = 7;
    }
    stepper_apply_phase(g_phase);
    delay(g_delay_ms);
    g_position--;
    steps++;
  }
}

void stepper_release() {
  digitalWrite(STEPPER_PIN_IN1, LOW);
  digitalWrite(STEPPER_PIN_IN2, LOW);
  digitalWrite(STEPPER_PIN_IN3, LOW);
  digitalWrite(STEPPER_PIN_IN4, LOW);
}
```
