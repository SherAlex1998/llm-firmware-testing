#include "logic.h"
#include "test_runner.h"

TEST(initialization_sets_all_yellow) {
  ti_init();

  for (uint8_t d = 0; d < TI_NUM_DIRECTIONS; d++) {
    ASSERT_EQ(ti_direction_color(d), TI_YELLOW);
  }
  ASSERT_EQ(ti_phase(), 0);
  ASSERT_TRUE(ti_is_safe());

  ASSERT_PIN(13, LOW);
  ASSERT_PIN(12, LOW);
  ASSERT_PIN(11, HIGH);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, LOW);
  ASSERT_PIN(8, HIGH);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, LOW);
  ASSERT_PIN(5, HIGH);
  ASSERT_PIN(4, LOW);
  ASSERT_PIN(3, LOW);
  ASSERT_PIN(2, HIGH);
}

TEST(set_direction_drives_selected_color) {
  ti_init();

  ASSERT_TRUE(ti_set_direction(1, TI_GREEN));
  ASSERT_EQ(ti_direction_color(1), TI_GREEN);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, HIGH);
  ASSERT_PIN(8, LOW);

  ASSERT_TRUE(ti_set_direction(1, TI_YELLOW));
  ASSERT_EQ(ti_direction_color(1), TI_YELLOW);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, LOW);
  ASSERT_PIN(8, HIGH);

  ASSERT_TRUE(ti_set_direction(1, TI_RED));
  ASSERT_EQ(ti_direction_color(1), TI_RED);
  ASSERT_PIN(10, HIGH);
  ASSERT_PIN(9, LOW);
  ASSERT_PIN(8, LOW);
}

TEST(invalid_direction_and_color_are_rejected) {
  ti_init();

  ASSERT_TRUE(ti_set_direction(2, TI_GREEN));
  ASSERT_FALSE(ti_set_direction(4, TI_RED));
  ASSERT_FALSE(ti_set_direction(2, 3));
  ASSERT_EQ(ti_direction_color(2), TI_GREEN);
  ASSERT_EQ(ti_direction_color(4), TI_RED);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, HIGH);
  ASSERT_PIN(5, LOW);
}

TEST(apply_phase_sets_one_active_direction) {
  ti_init();

  ASSERT_TRUE(ti_apply_phase(4));
  ASSERT_EQ(ti_phase(), 4);
  ASSERT_TRUE(ti_is_safe());
  ASSERT_EQ(ti_direction_color(0), TI_RED);
  ASSERT_EQ(ti_direction_color(1), TI_RED);
  ASSERT_EQ(ti_direction_color(2), TI_GREEN);
  ASSERT_EQ(ti_direction_color(3), TI_RED);
  ASSERT_PIN(13, HIGH);
  ASSERT_PIN(10, HIGH);
  ASSERT_PIN(7, LOW);
  ASSERT_PIN(6, HIGH);
  ASSERT_PIN(4, HIGH);

  ASSERT_TRUE(ti_apply_phase(7));
  ASSERT_EQ(ti_phase(), 7);
  ASSERT_EQ(ti_direction_color(3), TI_YELLOW);
  ASSERT_PIN(4, LOW);
  ASSERT_PIN(3, LOW);
  ASSERT_PIN(2, HIGH);
}

TEST(invalid_phase_preserves_current_phase) {
  ti_init();

  ASSERT_TRUE(ti_apply_phase(2));
  ASSERT_FALSE(ti_apply_phase(8));
  ASSERT_EQ(ti_phase(), 2);
  ASSERT_EQ(ti_direction_color(1), TI_GREEN);
  ASSERT_PIN(10, LOW);
  ASSERT_PIN(9, HIGH);
  ASSERT_PIN(8, LOW);
}

TEST(next_phase_advances_and_wraps) {
  ti_init();

  ti_next_phase();
  ASSERT_EQ(ti_phase(), 1);
  ASSERT_EQ(ti_direction_color(0), TI_YELLOW);
  ASSERT_EQ(ti_direction_color(1), TI_RED);
  ASSERT_PIN(11, HIGH);
  ASSERT_PIN(10, HIGH);

  ASSERT_TRUE(ti_apply_phase(7));
  ti_next_phase();
  ASSERT_EQ(ti_phase(), 0);
  ASSERT_EQ(ti_direction_color(0), TI_GREEN);
  ASSERT_EQ(ti_direction_color(3), TI_RED);
  ASSERT_PIN(12, HIGH);
  ASSERT_PIN(4, HIGH);
}

TEST(safety_detects_multiple_non_red_directions) {
  ti_init();

  ASSERT_TRUE(ti_set_direction(0, TI_GREEN));
  ASSERT_TRUE(ti_is_safe());

  ASSERT_TRUE(ti_set_direction(1, TI_YELLOW));
  ASSERT_FALSE(ti_is_safe());

  ASSERT_TRUE(ti_set_direction(1, TI_RED));
  ASSERT_TRUE(ti_is_safe());
}

void setup() { Serial.begin(9600); RUN_ALL_TESTS(); }
void loop() {}
