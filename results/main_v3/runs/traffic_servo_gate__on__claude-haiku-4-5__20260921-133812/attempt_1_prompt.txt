```cpp
// traffic_servo_gate: 3-LED traffic light whose green phase opens a servo barrier.
// Adapted from https://github.com/Rifat1003/Simple-Traffic-control-system-using-LED-and-servo-motor./tree/main (MIT).
#ifndef LOGIC_H
#define LOGIC_H
#include <Arduino.h>
#include <Servo.h>

#define TSG_PIN_GREEN  11
#define TSG_PIN_YELLOW 12
#define TSG_PIN_RED    13
#define TSG_PIN_SERVO  10
#define TSG_GATE_OPEN_ANGLE   180
#define TSG_GATE_CLOSED_ANGLE 0

enum TsgState : uint8_t {
  TSG_GREEN = 0,    // green LED, gate open
  TSG_YELLOW = 1,   // yellow LED, gate closed
  TSG_RED = 2,      // red LED, gate closed
  TSG_OFF = 3,      // all LEDs off, gate unchanged
  TSG_STATE_COUNT = 4
};

void tsg_init();                                    // pins configured, servo attached, enter TSG_GREEN
bool tsg_set_state(uint8_t state);                  // drive LEDs and servo for a state; false if invalid
uint8_t tsg_state();
uint8_t tsg_next_state(uint8_t state);              // pure: cyclic successor
void tsg_next();                                    // set_state(next_state(current))
int tsg_gate_angle();                               // Servo::read()
bool tsg_gate_open();                               // gate angle >= 90
unsigned int tsg_state_duration_ms(uint8_t state);  // pure: 5000 for every valid state

#endif
```

### logic.cpp
```cpp
#include "logic.h"

static Servo g_servo;
static uint8_t g_state = TSG_GREEN;

static void tsg_leds(bool green, bool yellow, bool red) {
  digitalWrite(TSG_PIN_GREEN, green ? HIGH : LOW);
  digitalWrite(TSG_PIN_YELLOW, yellow ? HIGH : LOW);
  digitalWrite(TSG_PIN_RED, red ? HIGH : LOW);
}

void tsg_init() {
  pinMode(TSG_PIN_GREEN, OUTPUT);
  pinMode(TSG_PIN_YELLOW, OUTPUT);
  pinMode(TSG_PIN_RED, OUTPUT);
  g_servo.attach(TSG_PIN_SERVO, 500, 2500);
  tsg_set_state(TSG_GREEN);
}

bool tsg_set_state(uint8_t state) {
  if (state >= TSG_STATE_COUNT) {
    return false;
  }
  g_state = state;
  switch (state) {
    case TSG_GREEN:
      tsg_leds(true, false, false);
      g_servo.write(TSG_GATE_OPEN_ANGLE);
      break;
    case TSG_YELLOW:
      tsg_leds(false, true, false);
      g_servo.write(TSG_GATE_CLOSED_ANGLE);
      break;
    case TSG_RED:
      tsg_leds(false, false, true);
      g_servo.write(TSG_GATE_CLOSED_ANGLE);
      break;
    case TSG_OFF:
      tsg_leds(false, false, false);
      break;
  }
  return true;
}

uint8_t tsg_state() {
  return g_state;
}

uint8_t tsg_next_state(uint8_t state) {
  if (state >= TSG_STATE_COUNT) {
    return TSG_GREEN;
  }
  return (state + 1) % TSG_STATE_COUNT;
}

void tsg_next() {
  tsg_set_state(tsg_next_state(g_state));
}

int tsg_gate_angle() {
  return g_servo.read();
}

bool tsg_gate_open() {
  return tsg_gate_angle() >= 90;
}

unsigned int tsg_state_duration_ms(uint8_t state) {
  if (state >= TSG_STATE_COUNT) {
    return 0;
  }
  return 5000;
}
```

## Covering a function
- Call the module's init function at the start of every test so that tests do not depend on each other.
- For every output a function drives, assert each pin it should set and each pin it should leave unchanged.
- For functions with an index or a range argument (LED number, state number), test the first value, the last
  value and one value outside the range.

## Servo library
- On the Uno the Servo library uses Timer1 and disables `analogWrite()` on pins 9 and 10, whether or not a servo
  is attached to them.
- `Servo::write(angle)` clamps angles below 0 to 0 and above 180 to 180; values of 544 and more are treated as
  a pulse width in microseconds (Servo 1.3.0, `Servo.cpp`).
- `Servo::read()` converts the last pulse width back to degrees, so it returns the last angle written; it does
  not measure the position of the shaft. After `attach()` and before any `write()` the pulse is 1500 us.
- A test for a servo module can call the function that moves the servo and check `read()` or the module's angle
  getter; for a sweep, check the first, the last and one intermediate angle, and that the angle never leaves
  the allowed range.

## Pin numbers on the Uno
- Digital pins 0 to 13. The analog inputs A0 to A5 are also digital pins 14 to 19.
- Pins 0 and 1 are the Serial RX/TX lines; the test protocol uses Serial, so tests must not drive them.
- PWM pins: 3, 5, 6, 9, 10, 11. `LED_BUILTIN` is pin 13.
- Check outputs against the pin numbers in the module's specification (its pin table), not against constants
  taken from the code under test: a wrong pin number in the code is only detected if the test uses the pin
  number the hardware is actually wired to.

# Testing PWM, analog inputs and servos

## analogWrite and what a test can observe
`analogWrite(pin, value)` sets the pin to `OUTPUT`. `analogWrite(pin, 0)` is implemented as
`digitalWrite(pin, LOW)` and `analogWrite(pin, 255)` as `digitalWrite(pin, HIGH)`; any other value starts the
PWM timer (Arduino AVR core, `wiring_analog.c`). On the Uno the PWM frequency is 490 Hz, and 980 Hz on pins 5
and 6.

Only the two extreme values can be checked with `digitalRead()`. For values 1 to 254 the duty cycle cannot be
read back, and `digitalRead()` would switch the PWM off. Check intermediate levels through the module's own
state: a getter that returns the level it wrote, or the value returned by the function that computes the level.

Servo library / Circuit

## Circuit

Servo motors have three wires: power, ground, and signal. The power wire is typically red, and should be connected to the 5V pin on the Arduino board. The ground wire is typically black or brown and should be connected to a ground pin on the Arduino board. The signal pin is typically yellow, orange or white and should be connected to a digital pin on the Arduino board. Note that servos draw considerable power, so if you need to drive more than one or two, you'll probably need to power them from a separate supply (i.e. not the 5V pin on your Arduino). Be sure to connect the grounds of the Arduino and external power supply together.
