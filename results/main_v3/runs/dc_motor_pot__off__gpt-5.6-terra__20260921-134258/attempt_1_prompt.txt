```cpp
// dc_motor_pot: DC motor via L298N; speed from a potentiometer (PWM on EN), direction on IN1/IN2.
// Adapted from https://github.com/AndreaLombardo/Arduino-L298N-Motore-DC (MIT).
#ifndef LOGIC_H
#define LOGIC_H
#include <Arduino.h>

#define MOTOR_PIN_POT A0    // potentiometer (analog)
#define MOTOR_PIN_IN1 12    // L298N IN1
#define MOTOR_PIN_IN2 13    // L298N IN2
#define MOTOR_PIN_EN  11    // L298N EN (PWM)

void motor_init();                          // pins configured, forward direction, power 0
void motor_set_direction(bool forward);     // forward: IN1 LOW / IN2 HIGH; reverse: IN1 HIGH / IN2 LOW
bool motor_direction();                     // true = forward
void motor_set_power(int pwm);              // clamped to 0..255, written to EN
int motor_power();                          // last PWM value written
int motor_power_from_reading(int raw);      // pure: ADC 0..1023 -> PWM 0..255
void motor_stop();                          // IN1 LOW, IN2 LOW, power 0
void motor_update();                        // read potentiometer and apply the power

#endif
```

### logic.cpp
```cpp
#include "logic.h"

static bool g_forward = true;
static int g_power = 0;

void motor_init() {
  pinMode(MOTOR_PIN_POT, INPUT);
  pinMode(MOTOR_PIN_IN1, OUTPUT);
  pinMode(MOTOR_PIN_IN2, OUTPUT);
  pinMode(MOTOR_PIN_EN, OUTPUT);
  motor_set_direction(true);
  motor_set_power(0);
}

void motor_set_direction(bool forward) {
  g_forward = forward;
  if (forward) {
    digitalWrite(MOTOR_PIN_IN1, LOW);
    digitalWrite(MOTOR_PIN_IN2, HIGH);
  } else {
    digitalWrite(MOTOR_PIN_IN1, HIGH);
    digitalWrite(MOTOR_PIN_IN2, LOW);
  }
}

bool motor_direction() {
  return g_forward;
}

void motor_set_power(int pwm) {
  if (pwm < 0) {
    pwm = 0;
  }
  if (pwm > 255) {
    pwm = 255;
  }
  g_power = pwm;
  analogWrite(MOTOR_PIN_EN, g_power);
}

int motor_power() {
  return g_power;
}

int motor_power_from_reading(int raw) {
  if (raw < 0) {
    raw = 0;
  }
  if (raw > 1023) {
    raw = 1023;
  }
  return map(raw, 0, 1023, 0, 255);
}

void motor_stop() {
  digitalWrite(MOTOR_PIN_IN1, LOW);
  digitalWrite(MOTOR_PIN_IN2, LOW);
  motor_set_power(0);
}

void motor_update() {
  int raw = analogRead(MOTOR_PIN_POT);
  motor_set_power(motor_power_from_reading(raw));
}
```
